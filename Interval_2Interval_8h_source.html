48 // Semitones of the perfect or major simple interval for 0-6 steps
49 // (0 2 4 5 7 9 11). Computed rather than tabled so AVR keeps no lookup
50 // table in RAM.
51 static constexpr int32_t ReferenceSemitones(int32_t simpleSteps) noexcept {
52 return simpleSteps * 2 - (simpleSteps >= 3 ? 1 : 0);
53 }
54
55 static constexpr bool IsPerfectSteps(int32_t simpleSteps) noexcept {
56 return simpleSteps == 0 || simpleSteps == 3 || simpleSteps == 4;
57 }
58
59 static constexpr int32_t Magnitude(int32_t value) noexcept {
60 return value < 0 ? -value : value;
61 }

63 // Deviation of the semitones from the perfect/major reference of the
64 // same number, measured in the interval's direction.
65 MCC_CONSTEXPR14 int32_t Deviation() const noexcept {
66 const int32_t sign = Sign();
67 const int32_t steps = sign * _steps;
68 const int32_t semitones = sign * _semitones;
69 return semitones - (steps / 7) * 12 - ReferenceSemitones(steps % 7);
70 }
71
72 constexpr int32_t Sign() const noexcept {
73 return (_steps != 0)
74 ? (_steps > 0 ? 1 : -1)
75 : (_semitones < 0 ? -1 : 1);
76 }
77
78 static MCC_CONSTEXPR14 IntervalQuality QualityFor(int32_t steps,
79 int32_t deviation) noexcept {
80 if (IsPerfectSteps(steps % 7)) {
81 if (deviation == 0) {
82 return IntervalQuality::Perfect();
83 }
84 return deviation > 0
85 ? IntervalQuality::Augmented(deviation)
86 : IntervalQuality::Diminished(-deviation);
87 }
88 if (deviation == 0) {
89 return IntervalQuality::Major();
90 }
91 if (deviation == -1) {
92 return IntervalQuality::Minor();
93 }
94 return deviation > 0
95 ? IntervalQuality::Augmented(deviation)
96 : IntervalQuality::Diminished(-deviation - 1);
97 }

111 static MCC_CONSTEXPR14 Interval FromSteps(int32_t diatonicSteps,
112 int32_t semitones) noexcept {
113 // Compare against both bounds: negating INT32_MIN would overflow.
114 if (diatonicSteps < -MaximumSteps || diatonicSteps > MaximumSteps ||
115 semitones < -MaximumSemitones || semitones > MaximumSemitones) {
116 return Interval();
117 }
118 Interval candidate;
119 candidate._steps = static_cast<int16_t>(diatonicSteps);
120 candidate._semitones = static_cast<int16_t>(semitones);
121 return candidate.Quality().IsValid() ? candidate : Interval();
122 }

198 MCC_CONSTEXPR14 Interval Simple() const noexcept {
199 if (!IsValid() || Magnitude(_steps) <= 7) {
200 return *this;
201 }
202 const int32_t sign = Sign();
203 const int32_t octaves = (Magnitude(_steps) - 1) / 7;
204 return FromSteps(_steps - sign * octaves * 7,
205 _semitones - sign * octaves * 12);
206 }

217 MCC_CONSTEXPR14 Interval Inverted() const noexcept {
218 if (!IsValid()) {

222 const int32_t sign = simple.Sign();
223 return FromSteps(sign * 7 - simple._steps, sign * 12 - simple._semitones);
224 }

262 static MCC_CONSTEXPR14 Interval Build(IntervalQuality quality, IntervalNumber number,
263 IntervalDirection direction) noexcept {
264 if (!quality.IsValid() || !number.IsValid()) {
265 return Interval();
266 }
267 const int32_t steps = number.Value() - 1;
268 const int32_t simpleSteps = steps % 7;
269 const IntervalQualityKind kind = quality.Kind();
270 const int32_t count = quality.Count();
271
272 int32_t deviation = 0;
273 if (IsPerfectSteps(simpleSteps)) {
274 if (kind == IntervalQualityKind::Major ||
275 kind == IntervalQualityKind::Minor) {
276 return Interval();
277 }
278 deviation = (kind == IntervalQualityKind::Augmented) ? count
279 : (kind == IntervalQualityKind::Diminished) ? -count
280 : 0;
281 } else {
282 if (kind == IntervalQualityKind::Perfect) {
283 return Interval();
284 }
285 deviation = (kind == IntervalQualityKind::Augmented) ? count
286 : (kind == IntervalQualityKind::Minor) ? -1
287 : (kind == IntervalQualityKind::Diminished) ? -count - 1
288 : 0;
289 }
290
291 if (steps == 0) {
292 if (deviation < 0) {
293 return Interval(); // No diminished unison (SPEC-INT-5).
294 }
295 if (deviation == 0) {
296 return Unison();
297 }
298 if (direction == IntervalDirection::Unison) {
299 return Interval();
300 }
301 } else if (direction == IntervalDirection::Unison) {
302 return Interval();
303 }
304
305 const int32_t sign = (direction == IntervalDirection::Descending) ? -1 : 1;
306 const int32_t semitones =
307 (steps / 7) * 12 + ReferenceSemitones(simpleSteps) + deviation;
308 return FromSteps(sign * steps, sign * semitones);
309 }
