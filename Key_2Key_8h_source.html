50 // Circle-of-fifths position of a letter relative to C: F -1 ... B 5.
51 static constexpr int32_t LetterFifths(Letter letter) noexcept {
52 return (DiatonicIndex(letter) * 2 + 1) % 7 - 1;

55 // Signature offset of a mode relative to major on the same tonic.

80 static MCC_CONSTEXPR14 Key FromSignature(KeySignature signature, KeyMode mode) noexcept {
81 if (!signature.IsValid() || !IsMode(mode)) {
82 return Key();
83 }
84 // Tonic position on the circle of fifths, then its letter and accidental.
85 const int32_t fifths = signature.Fifths() - ModeOffset(mode);
86 const int32_t order = Foundation::Math::FloorMod(fifths + 1, 7);
87 const int32_t accidental = Foundation::Math::FloorDiv(fifths + 1, 7);
88 const Letter letter = static_cast<Letter>(((order + 6) * 4) % 7);
89 return Key(NoteName(letter, Accidental(accidental)), mode);
90 }

116 MCC_CONSTEXPR14 NoteName Spell(MCC::PitchClass pitchClass) const noexcept {
117 if (!IsValid() || !pitchClass.IsValid()) {
118 return NoteName::Invalid();
119 }
120 const KeySignature signature = Signature();
121 const int32_t direction = signature.Fifths() < 0 ? -1 : 1;
122 NoteName best;
123 int32_t bestScore = 1000;
124 for (int32_t index = 0; index < 7; ++index) {
125 const Letter letter = static_cast<Letter>(index);
126 const int32_t keyAccidental = signature.AccidentalOf(letter).Semitones();
127 for (int32_t alteration = -1; alteration <= 1; ++alteration) {
128 const NoteName candidate(letter, Accidental(keyAccidental + alteration));
129 if (candidate.IsValid() && candidate.PitchClass() == pitchClass) {
130 // Diatonic first, then the fewest accidentals, then the
131 // key's direction: raised in sharp keys, lowered in flat keys.
132 const int32_t accidental = candidate.Accidental().Semitones();
133 const int32_t score = alteration == 0 ? 0
134 : 10 * (accidental < 0 ? -accidental : accidental) +
135 (alteration == direction ? 1 : 2);
136 if (score < bestScore) {
137 best = candidate;
138 bestScore = score;
139 }
140 }
141 }
142 }
143 return best;
144 }

147 MCC_CONSTEXPR14 MCC::Pitch Spell(MCC::ChromaticIndex index) const noexcept {
148 if (!index.IsValid()) {
149 return MCC::Pitch::Invalid();
150 }
151 const NoteName noteName = Spell(index.PitchClass());
152 if (!noteName.IsValid()) {
153 return MCC::Pitch::Invalid();
154 }
155 const int32_t offset =
156 NaturalSemitone(noteName.Letter()) + noteName.Accidental().Semitones();
157 return MCC::Pitch(noteName, Foundation::Math::FloorDiv(index.Value() - offset, 12) - 1);
158 }
