110 size_t Voicing(int32_t inversion, int32_t rootOctave,
111 MCC::Pitch* pitches, size_t capacity) const noexcept {
112 const uint8_t count = ToneCount();
113 if (count == 0 || inversion < 0 || inversion >= count) {
114 return 0;
115 }
116 MCC::Pitch previous;
117 for (uint8_t i = 0; i < count; ++i) {
118 const int32_t tone = (inversion + i) % count + 1;
119 MCC::Pitch pitch = PitchAt(tone, rootOctave);
120 while (previous.IsValid() && pitch.IsValid() && !IsLowerThan(previous, pitch)) {
121 pitch = pitch.MovedByOctaves(1);
122 }
123 if (i < capacity) {
124 pitches[i] = pitch;
125 }
126 previous = pitch;
127 }
128 return count;
