47MCC_CONSTEXPR14 Interval IntervalBetween(NoteName from, NoteName to) noexcept {
48 if (!from.IsValid() || !to.IsValid()) {
49 return Interval::Invalid();
50 }
51 const int32_t fromIndex = DiatonicIndex(from.Letter());
52 const int32_t toIndex = DiatonicIndex(to.Letter());
53 const int32_t steps = Foundation::Math::FloorMod(toIndex - fromIndex, LetterCount);
54 const int32_t wrap = (toIndex < fromIndex) ? 12 : 0;
55 const int32_t semitones =
56 NaturalSemitone(to.Letter()) + to.Accidental().Semitones() + wrap -
57 NaturalSemitone(from.Letter()) - from.Accidental().Semitones();
58 return Interval::FromSteps(steps, semitones);
59}

71MCC_CONSTEXPR14 Pitch operator+(Pitch pitch, Interval interval) noexcept {
72 if (!pitch.IsValid() || !interval.IsValid()) {
73 return Pitch::Invalid();
74 }
75 const int32_t diatonic =
76 static_cast<int32_t>(pitch.DiatonicIndex()) + interval.DiatonicSteps();
77 const int32_t chromatic =
78 static_cast<int32_t>(pitch.ChromaticIndex().Value()) + interval.Semitones();
79 const Letter letter =
80 static_cast<Letter>(Foundation::Math::FloorMod(diatonic, LetterCount));
81 const int32_t octave = Foundation::Math::FloorDiv(diatonic, LetterCount);
82 const int32_t natural = (octave + 1) * 12 + NaturalSemitone(letter);
83 return Pitch(letter, Accidental(chromatic - natural), octave);
84}

103MCC_CONSTEXPR14 NoteName operator+(NoteName noteName, Interval interval) noexcept {
104 if (!noteName.IsValid() || !interval.IsValid()) {
105 return NoteName::Invalid();

107 // Write the note name in octave 0, transpose it by the simple interval
108 // (at most one octave, so the octave stays in range) and drop the
109 // octave. An accidental overflow yields the invalid pitch, whose note
110 // name is invalid.
