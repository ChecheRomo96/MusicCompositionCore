78 const int32_t distance =
79 static_cast<int32_t>(pitch.ChromaticIndex().Value()) -
80 _tuning.Reference().ChromaticIndex().Value();
81 int32_t octaves = Foundation::Math::FloorDiv(
82 distance, MCC::PitchClass::Count);
83 const int32_t semitones = Foundation::Math::FloorMod(
84 distance, MCC::PitchClass::Count);
85
86 float frequency = _tuning.ReferenceFrequency() * ratios[semitones];
87 for (; octaves > 0; --octaves) {
88 if (frequency > FLT_MAX / 2.0f) {
89 return 0.0f;
90 }
91 frequency *= 2.0f;
92 }
93 for (; octaves < 0; ++octaves) {
94 frequency *= 0.5f;
95 }
