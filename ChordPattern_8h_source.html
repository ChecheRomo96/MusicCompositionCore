49 // Semitones of the major-scale degree `steps` above the root, extended
50 // past the octave. Computed rather than tabled so AVR keeps no RAM table.
51 static constexpr int32_t NaturalSemitones(int32_t steps) noexcept {
52 return (steps / 7) * 12 + (steps % 7) * 2 - ((steps % 7) >= 3 ? 1 : 0);
53 }
54
55 constexpr uint8_t StepsAt(uint8_t index) const noexcept {
56 return static_cast<uint8_t>((_steps[index / 2] >> ((index % 2) * 4)) & 0x0F);
57 }

59 MCC_CONSTEXPR14 int32_t SemitonesAt(uint8_t index) const noexcept {
60 uint8_t seen = 0;
61 for (int32_t k = 0; k < 24; ++k) {
62 if ((_semitones >> k) & 1u) {
63 if (seen == index) {
64 return k;
65 }
66 ++seen;
67 }
68 }
69 return -1;
70 }

88 static MCC_CONSTEXPR14 ChordPattern FromFormula(const char* formula) noexcept {
89 ChordPattern pattern;
90 if (formula == nullptr) {
91 return Invalid();
92 }
93 uint8_t count = 0;
94 int32_t previous = -1;
95 // Index-based scanning without `continue` keeps GCC 7 (the Arduino
96 // AVR toolchain) able to evaluate the loop at compile time.
97 int32_t i = 0;
98 while (formula[i] != '\0') {
99 if (formula[i] == ' ') {
100 ++i;
101 } else {
102 int32_t accidental = 0;
103 while (formula[i] == 'b' || formula[i] == '#') {
104 accidental += (formula[i] == '#') ? 1 : -1;
105 ++i;
106 }
107 int32_t number = 0;
108 while (formula[i] >= '0' && formula[i] <= '9' && number < 100) {
109 number = number * 10 + (formula[i] - '0');
110 ++i;
111 }
112 if (number < 1 || number > 13 ||
113 (formula[i] != ' ' && formula[i] != '\0')) {
114 return Invalid();
115 }
116 const int32_t steps = number - 1;
117 const int32_t semitones = NaturalSemitones(steps) + accidental;
118 if (count == MaximumTones || semitones <= previous || semitones > 23 ||
119 (count == 0 && (steps != 0 || semitones != 0)) ||
120 !Interval::FromSteps(steps, semitones).IsValid()) {
121 return Invalid();
122 }
123 pattern._semitones |= (static_cast<uint32_t>(1) << semitones);
124 pattern._steps[count / 2] = static_cast<uint8_t>(
125 pattern._steps[count / 2] | (steps << ((count % 2) * 4)));
126 previous = semitones;
127 ++count;
128 }
129 }
130 return count == 0 ? Invalid() : pattern;
131 }
